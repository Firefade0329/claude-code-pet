import { test, expect, mock } from 'claude-code/testing'

// Probes for the round-6 review. Each test asserts what I EXPECT the current code to do;
// a pass means the behaviour described in its name is real in the test environment.
const PROPS = { hasSurvey: false, isWorking: false, maxRows: 20, bodyColumns: 120, scroll: { offset: 0, bodyRows: 20 }, view: {} } as any
const mount = ($: any) => $.ui.mount({ plugin: 'pet', surface: 'desktop', component: 'AbovePrompt', props: PROPS })
let gates: Record<string, Promise<void>> = {}
async function engine($: any, on: any, entries?: Record<string, unknown>) {
  const clock = mock.clock(on)
  mock.store(on, entries)
  on('ui.toast', () => ({ value: undefined } as any))
  on('fs.read', () => ({ value: { base64: 'AAAA' } } as any))
  on('session.start', () => ({ cwd: '/test' } as any))
  on('turn.start', (_: any, e: any) => ({ turnId: e.turnId } as any))
  on('turn.complete', () => ({ text: '' } as any))
  on('tool.call', async (_: any, e: any) => {
    if (gates[e.tool]) await gates[e.tool]
    return { result: 'ok' } as any
  })
  on('classic.PermissionRequest', () => ({}))
  await clock.set(new Date(2023, 10, 15, 15, 0).getTime())      // 15:00 local: no night mood
  return clock
}
const count = async (m: any, text: RegExp | string) => (await m.findAll({ type: 'Text', text })).length
const countBtn = async (m: any, text: RegExp | string) => (await m.findAll({ type: 'Button', text })).length
const faces = async (m: any) => (await m.findAll({ type: 'Svg' })).map((n: any) => n.props.alt).filter((a: string) => a.startsWith('pet: '))
const texts = async (m: any) => (await m.findAll({ type: 'Text' })).map((n: any) => String(n.props.children ?? n.text ?? '')).join(' | ')
const callTool = ($: any, tool: string, args: Record<string, unknown> = {}) => $.tool.call({ tool, tool_use_id: 'k' + Math.random(), ...args } as any)
let agentSeq = 0
const withAgents = (on: any) => on('agent.spawn', () => ({ model: 'm', agentId: 'ag' + ++agentSeq } as any))
const spawn = ($: any) => $.agent.spawn({ prompt: 'p', description: 'd' } as any)
const endAgent = ($: any, id: string) => $.turn.complete({ reason: 'answer', answer: '', durationMs: 1000, isAborted: false, turnId: 'x' + id, agentId: id } as any)
const finishTurn = async ($: any, clock: any, id: string) => {
  await $.turn.start({ text: 'hi', turnId: id } as any)
  await clock.advance(1000)
  await $.turn.complete({ reason: 'answer', answer: '', durationMs: 1000, isAborted: false, turnId: id } as any)
  await clock.advance(8000)
}
const READ_LINE = /查阅资料|翻阅相关文件|仔细阅读|逐份过目|看明白/
const RUN_LINE = /执行命令|正在运行，请稍等|命令执行中|运行程序|正在验证/
const WATCH_LINE = /等候汇报|尚在处理|等候结果|等待助手们|仍在办理|汇报|守着|没回来|做得怎么样|等消息|等他们|一有消息|一起等/

// P1: background agents keep working after her own turn ended (Agent in the background, or a Workflow run)
test('P1a: after the main turn, a background agent\'s first tool call turns her into a typist (no watch)', async ($, on) => {
  agentSeq = 0
  const clock = await engine($, on)
  withAgents(on)
  const m = await mount($)
  await $.session.start({ source: 'startup', cwd: '/test' } as any)
  await clock.advance(15000)
  await finishTurn($, clock, 'main1')
  await spawn($)
  await spawn($)
  await clock.advance(20000)
  expect(await faces(m)).toEqual(['pet: watch'])      // as the 0.6.0 test shows
  await $.turn.start({ text: 'p', turnId: 'a1', agentId: 'ag1' } as any)
  await callTool($, 'Read', { file_path: '/x', agentId: 'ag1' })
  await clock.advance(1000)
  console.log('P1a faces after agent tool call:', JSON.stringify(await faces(m)))
  expect(await faces(m)).toEqual(['pet: work'])
})

// P3: in a long run whose tool kind changes now and then, how often is the long-run line shown?
test('P3: long run, the kind changes every 2 minutes: share of minutes 21-60 that show the long-run line', async ($, on) => {
  const clock = await engine($, on)
  const m = await mount($)
  await $.session.start({ source: 'startup', cwd: '/test' } as any)
  await $.turn.start({ text: 'big', turnId: 'L' } as any)
  await clock.advance(15000)
  const tools = ['Read', 'Bash', 'Grep', 'Edit']
  let longCount = 0
  let samples = 0
  for (let minute = 1; minute <= 60; minute++) {
    if (minute % 2 === 0) await callTool($, tools[(minute / 2) % 4], {})
    await clock.advance(60 * 1000)
    if (minute > 21) {
      samples += 1
      if ((await count(m, /\d+ 分钟|小时/)) > 0) longCount += 1
    }
  }
  console.log('P3 long-run line shown in', longCount, 'of', samples, 'sampled minutes')
  expect(longCount).toBe(samples)
})

test('P7: a background agent after her turn keeps the stretch of work alive through a 20 minute absence: the reminder comes right after the person returns', async ($, on) => {
  agentSeq = 0
  const clock = await engine($, on)
  withAgents(on)
  const m = await mount($)
  await $.session.start({ source: 'startup', cwd: '/test' } as any)
  await $.turn.start({ text: 'start', turnId: 'm0' } as any)      // T0: a stretch of work starts, reminder due at T0+60
  for (let i = 0; i < 3; i++) { await clock.advance(10 * 60 * 1000); if (i === 2) await $.ui.press({ plugin: 'pet', key: 'pat' }) }      // T0+30: the person is there (a pat), so the 45-minute rule does not apply below
  await $.turn.complete({ reason: 'answer', answer: '', durationMs: 30 * 60000, isAborted: false, turnId: 'm0' } as any)      // T0+30
  await spawn($)
  await $.turn.start({ text: 'p', turnId: 'a1', agentId: 'ag1' } as any)
  await callTool($, 'Read', { file_path: '/x', agentId: 'ag1' })
  await clock.advance(60 * 1000)
  await endAgent($, 'ag1')      // T0+31: nothing runs any more
  for (let i = 0; i < 4; i++) await clock.advance(5 * 60 * 1000)      // T0+51: twenty quiet minutes (more than the 15 minute gap)
  await $.turn.start({ text: 'back', turnId: 'm1' } as any)
  await clock.advance(10 * 60 * 1000 + 30000)      // T0+61.5
  const shown = await count(m, REST)
  console.log('P7 reminder 10 minutes after coming back from a 20 minute break:', shown)
  expect(shown).toBe(1)
})

test('P8: agents that start and end at the same moment: does the helper count stay right?', async ($, on) => {
  agentSeq = 0
  const clock = await engine($, on)
  withAgents(on)
  const m = await mount($)
  await $.turn.start({ text: 'go', turnId: 's' } as any)
  await clock.advance(15000)
  const base = await imagesIn(m)
  const bad: string[] = []
  await spawn($); await spawn($)      // ag1 ag2
  await Promise.all([endAgent($, 'ag1'), endAgent($, 'ag2')])
  if ((await imagesIn(m)) !== base) bad.push('A ' + ((await imagesIn(m)) - base))
  await spawn($); await spawn($)      // ag3 ag4
  await Promise.all([spawn($), endAgent($, 'ag3')])      // ag5 in, ag3 out: 2 left
  if ((await imagesIn(m)) !== base + 2) bad.push('B ' + ((await imagesIn(m)) - base))
  await Promise.all([endAgent($, 'ag4'), spawn($), endAgent($, 'ag5')])      // ag6 in: 1 left
  if ((await imagesIn(m)) !== base + 1) bad.push('C ' + ((await imagesIn(m)) - base))
  console.log('P8 mismatches:', JSON.stringify(bad))
  expect(true).toBe(true)
})
